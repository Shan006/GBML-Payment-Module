Modular contract classes

pragma solidity ^0.8.0;

import "@openzeppelin/contracts/token/ERC20/ERC20.sol";

interface IJvdEgcrRouter {
    function route(address from, address to, uint256 amount, address token) external returns (bool);
}

contract JRC20WithJvdRouter is ERC20 {
    address public router;

    constructor(
        string memory name,
        string memory symbol,
        uint256 supply,
        address routerAddress
    ) ERC20(name, symbol) {
        _mint(msg.sender, supply);
        router = routerAddress;
    }

    function transfer(address to, uint256 amount) public override returns (bool) {
        require(IJvdEgcrRouter(router).route(msg.sender, to, amount, address(this)), "Routing failed");
        return super.transfer(to, amount);
    }

    function transferFrom(address from, address to, uint256 amount) public override returns (bool) {
        require(IJvdEgcrRouter(router).route(from, to, amount, address(this)), "Routing failed");
        return super.transferFrom(from, to, amount);
    }
}

🔧 /utils/deployer.js (Updated for JVD EGCR Routing)
import fs from 'fs'
import path from 'path'
import { ethers } from 'ethers'
import dotenv from 'dotenv'
dotenv.config()

const CONTRACTS_PATH = path.join(process.cwd(), 'contracts/templates')
const RPC_URL = process.env.RPC_URL
const PRIVATE_KEY = process.env.PRIVATE_KEY
const JVD_ROUTER_ADDRESS = process.env.JVD_ROUTER_ADDRESS // Deployed router address

const provider = new ethers.JsonRpcProvider(RPC_URL)
const wallet = new ethers.Wallet(PRIVATE_KEY, provider)

export async function compileAndDeploy(contractName, constructorArgs = []) {
  const artifact = JSON.parse(fs.readFileSync(`${CONTRACTS_PATH}/${contractName}.json`))
  const factory = new ethers.ContractFactory(artifact.abi, artifact.bytecode, wallet)

  const contract = await factory.deploy(...constructorArgs)
  await contract.waitForDeployment()

  return {
    abi: artifact.abi,
    address: contract.target
  }
}

import { compileAndDeploy } from '../utils/deployer.js'
import { updateDashboard } from '../config/dashboardWriter.js'
import dotenv from 'dotenv'
dotenv.config()

export default async function handler(req, res) {
  const { serviceName, tokenName, symbol, initialSupply } = req.body

  if (!serviceName || !tokenName || !symbol || !initialSupply) {
    return res.status(400).json({ error: 'Missing required fields' })
  }

  try {
    const routerAddress = process.env.JVD_ROUTER_ADDRESS

    // 3. Return info
    return res.status(200).json({
      message: `Tokenization for "${serviceName}" with enforced JVD EGCR routing successful`,
      contractAddress: address,
      abi,
      symbol,
      initialSupply
    })

  } catch (err) {
    console.error('Tokenization error:', err)
    return res.status(500).json({ error: 'Token deployment failed' })
  }
}

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

// Juvidoe Token Templates with JVD EGCR Routing
import "../templates/JRC20WithJvdRouter.sol";
import "../templates/JRC721WithJvdRouter.sol";
import "../templates/JRC998WithJvdRouter.sol";

contract TokenFactory {
    address public superAdmin;
    address public jvdEgcrRouter;

    event TokenDeployed(address indexed deployedAddress, string symbol, string tokenType, string serviceModule);

    modifier onlySuperAdmin() {
        require(msg.sender == superAdmin, "Not authorized");
        _;
    }

    constructor(address _router) {
        superAdmin = msg.sender;
        jvdEgcrRouter = _router;
    }

    function deployJRC20(
        string memory name,
        string memory symbol,
        uint256 totalSupply,
        string memory serviceModule
    ) external onlySuperAdmin returns (address) {
        JRC20WithJvdRouter token = new JRC20WithJvdRouter(name, symbol, totalSupply, jvdEgcrRouter);
        emit TokenDeployed(address(token), symbol, "JRC-20", serviceModule);
        return address(token);
    }

    function deployJRC721(
        string memory name,
        string memory symbol,
        string memory serviceModule
    ) external onlySuperAdmin returns (address) {
        JRC721WithJvdRouter token = new JRC721WithJvdRouter(name, symbol, jvdEgcrRouter);
        emit TokenDeployed(address(token), symbol, "JRC-721", serviceModule);
        return address(token);
    }

    function deployJRC998(
        string memory name,
        string memory symbol,
        string memory serviceModule
    ) external onlySuperAdmin returns (address) {
        JRC998WithJvdRouter token = new JRC998WithJvdRouter(name, symbol, jvdEgcrRouter);
        emit TokenDeployed(address(token), symbol, "JRC-998", serviceModule);
        return address(token);
    }

    function updateRouter(address newRouter) external onlySuperAdmin {
        require(newRouter != address(0), "Invalid router");
        jvdEgcrRouter = newRouter;
    }

    function transferSuperAdmin(address newAdmin) external onlySuperAdmin {
        require(newAdmin != address(0), "Invalid admin");
        superAdmin = newAdmin;
    }
}

const walletRequests = new Map()

export function walletRateLimiter(req, res, next) {
  const wallet = req.body.fromWallet
  const now = Date.now()

  const record = walletRequests.get(wallet) || { count: 0, last: now }

  if (now - record.last < 5 * 60 * 1000) { // 5 min cooldown
    if (record.count >= 2) {
      return res.status(429).json({ error: "Wallet tokenization rate exceeded." })
    }
    record.count += 1
  } else {
    record.count = 1
    record.last = now
  }

  walletRequests.set(wallet, record)
  next()
}

🔑 JvdPayWidget.js (Core Component)
import React, { useState } from 'react';
import { View, Text, Button, Picker, ActivityIndicator } from 'react-native';
import { initiatePayment } from './api/initiatePayment';
import { pollStatus } from './api/pollStatus';
import QRCodeBox from './ui/QRCodeBox';

export default function JvdPayWidget({ orderId, amountJvd, onSuccess }) {
  const [asset, setAsset] = useState("JVD");
  const [status, setStatus] = useState("idle");
  const [qr, setQr] = useState(null);

  const handlePay = async () => {
    setStatus("paying");
    try {
      const res = await initiatePayment(orderId, asset, amountJvd);
      if (res.qr) setQr(res.qr);
      if (res.redirectUrl) {
        Linking.openURL(res.redirectUrl);
        setStatus("redirected");
      }

      pollStatus(res.paymentId, (state, txHash) => {
        if (state === "settled") {
          setStatus("done");
          onSuccess({ txHash });
        }
      });
    } catch (e) {
      console.error(e);
      setStatus("error");
    }
  };

  return (
    <View style={{ padding: 16 }}>
      <Text style={{ fontSize: 16, marginBottom: 8 }}>Pay with:</Text>
      <Picker selectedValue={asset} onValueChange={(val) => setAsset(val)}>
        <Picker.Item label="JVD EGCR" value="JVD" />
        <Picker.Item label="ETH" value="ETH" />
        <Picker.Item label="USDT" value="USDT" />
        <Picker.Item label="Credit/Debit (Fiat)" value="FIAT" />
      </Picker>
      <Button title={`Pay ${amountJvd} JVD`} onPress={handlePay} />
      {status === "paying" && <ActivityIndicator style={{ marginTop: 16 }} />}
      {qr && <QRCodeBox qr={qr} />}
    </View>
  );
}

🧪 initiatePayment.js
export async function initiatePayment(orderId, payAsset, amountJvd) {
  const res = await fetch("https://your-backend/api/payments/initiate", {
    method: "POST",
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ orderId, payAsset, amountJvd })
  });
  return res.json();
}

🔁 pollStatus.js
export async function pollStatus(paymentId, onSettle) {
  const interval = setInterval(async () => {
    const res = await fetch(`https://your-backend/api/payments/status/${paymentId}`);
    const { state, txHash } = await res.json();
    if (state === "settled") {
      clearInterval(interval);
      onSettle(state, txHash);
    }
  }, 5000);
}

🔐 4. Signing + Routing Through JvdEgcrRouter
Once a wallet is connected, transactions should always go through your JvdEgcrRouter.sol.
import { ethers } from "ethers"
import JvdRouterABI from "./abis/JvdEgcrRouter.json"

const ROUTER_ADDRESS = "0xYourRouter"

export async function settleTransaction(walletProvider, to, jvdAmount, orderId) {
  const provider = new ethers.providers.Web3Provider(walletProvider)
  const signer = provider.getSigner()
  const router = new ethers.Contract(ROUTER_ADDRESS, JvdRouterABI, signer)

  const tx = await router.settleWithJvdEgcr(
    to,
    ethers.utils.parseUnits(jvdAmount.toString(), 18),
    orderId
  )

  return await tx.wait()
}
